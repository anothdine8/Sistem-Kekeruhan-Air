const int SCENARIO = 1;

unsigned long waktuMulai;
unsigned long waktuSelesai;

String getCurrentTime() {
  struct tm timeinfo;

  if (!getLocalTime(&timeinfo)) {
    return "Waktu tidak tersedia";
  }

  char buffer[25];

  strftime(buffer, sizeof(buffer),
           "%Y-%m-%d %H:%M:%S", &timeinfo);

  return String(buffer);
}

// ================= PENGUJIAN HTTP =================
void runScenario() {

  HTTPClient http;

  berhasil = 0;
  gagal = 0;

  waktuMulai = millis();

  String waktuMulaiPengujian = getCurrentTime();

  Serial.println("\n========================================");
  Serial.println("       PENGUJIAN HTTP KEEP-ALIVE");
  Serial.println("========================================");

  Serial.print("Skenario       : ");
  Serial.println(SCENARIO);

  Serial.print("Jumlah pesan   : ");
  Serial.println(TOTAL_PESAN);

  Serial.print("Payload        : ");
  Serial.print(PAYLOAD_SIZE);
  Serial.println(" byte");

  Serial.print("Content-Length : ");
  Serial.println(PAYLOAD_SIZE);

  Serial.print("Interval       : ");
  Serial.print(INTERVAL);
  Serial.println(" ms");

  Serial.print("Timeout        : ");
  Serial.print(TIMEOUT);
  Serial.println(" ms");

  Serial.print("Waktu mulai    : ");
  Serial.println(waktuMulaiPengujian);

  // ================= HTTP BEGIN =================
  http.begin(serverURL);

  http.setReuse(true);
  http.setConnectTimeout(TIMEOUT);
  http.setTimeout(TIMEOUT);

  http.addHeader("Content-Type", "text/plain");
  http.addHeader("Connection", "keep-alive");

  printHeader();

  // ================= LOOP PENGUJIAN =================
  for (int seq = 1; seq <= TOTAL_PESAN; seq++) {

    // Menentukan waktu pengiriman
    unsigned long targetTime =
      waktuMulai + ((unsigned long)(seq - 1) * INTERVAL);

    while ((long)(millis() - targetTime) < 0) {
      delay(1);
    }

    // ================= WAKTU MULAI =================
    String waktuMulaiPesan = getCurrentTime();

    unsigned long tKirim = millis() - waktuMulai;

    // ================= PAYLOAD 20 BYTE =================
    String body = "";

    for (int i = 0; i < PAYLOAD_SIZE; i++) {
      body += "A";
    }

    int contentLength = body.length();

    // ================= HTTP REQUEST =================
    int httpCode = -1;

    unsigned long tTerima = 0;
    unsigned long rtt = 0;

    String terimaStr = "-";
    String rttStr = "-";

    // ================= PENGIRIMAN =================
    if (WiFi.status() == WL_CONNECTED) {

      unsigned long requestStart = millis();

      httpCode = http.POST(
        (uint8_t*)body.c_str(),
        contentLength
      );

      // Membaca seluruh respons server
      String response = "";

      if (httpCode > 0) {
        response = http.getString();
      }

      unsigned long requestEnd = millis();

      rtt = requestEnd - requestStart;

      if (httpCode >= 200 && httpCode < 300) {

        tTerima = requestEnd - waktuMulai;

        terimaStr = String(tTerima);
        rttStr = String(rtt);

        berhasil++;

      } else {

        gagal++;
      }

    } else {

      gagal++;
    }

    // ================= STATUS =================
    int status =
      (httpCode >= 200 && httpCode < 300) ? 1 : 0;

    // ================= RSSI =================
    int rssi = WiFi.RSSI();

    // ================= OUTPUT TABEL =================
    Serial.printf(
      "%-8d %-6d %-20s %-8d %-9lu %-10s %-5s %-8d %d\n",
      SCENARIO,
      seq,
      waktuMulaiPesan.c_str(),
      contentLength,
      tKirim,
      terimaStr.c_str(),
      rttStr.c_str(),
      status,
      rssi
    );
  }

  // ================= AKHIR PENGUJIAN =================
  http.end();

  waktuSelesai = millis();

  String waktuAkhirPengujian = getCurrentTime();

  // ================= REKAP =================
  float pdr =
    (float)berhasil / TOTAL_PESAN * 100.0;

  Serial.println();
  Serial.println("========================================");
  Serial.println("           HASIL PENGUJIAN");
  Serial.println("========================================");

  Serial.print("Skenario             : ");
  Serial.println(SCENARIO);

  Serial.print("Waktu mulai          : ");
  Serial.println(waktuMulaiPengujian);

  Serial.print("Waktu selesai        : ");
  Serial.println(waktuAkhirPengujian);

  Serial.print("Total pengujian      : ");
  Serial.println(TOTAL_PESAN);

  Serial.print("Payload              : ");
  Serial.println(PAYLOAD_SIZE);

  Serial.print("Content-Length       : ");
  Serial.println(PAYLOAD_SIZE);

  Serial.print("Berhasil             : ");
  Serial.println(berhasil);

  Serial.print("Gagal                : ");
  Serial.println(gagal);

  Serial.print("PDR                  : ");
  Serial.print(pdr, 2);
  Serial.println("%");

  Serial.print("Durasi pengujian     : ");
  Serial.print(waktuSelesai - waktuMulai);
  Serial.println(" ms");

  Serial.println("========================================");
}

#include <WiFi.h>
#include <MQTT.h>      // library "MQTT" oleh Joel Gaehwiler (256dpi)
#include <time.h>

// ================= WIFI =================
const char* ssid = "Wokwi-GUEST";
const char* password = "";

// ================= MQTT =================
const char* MQTT_BROKER = "broker.emqx.io";
const int   MQTT_PORT   = 1883;

// ================= PARAMETER PENGUJIAN =================
// S3 = QoS0 20B | S4 = QoS0 500B | S5 = QoS1 20B | S6 = QoS1 500B
const int SCENARIO = 3;

const int TOTAL_PESAN = 100;
const unsigned long INTERVAL = 2000;
const unsigned long TIMEOUT  = 5000;

const int QOS          = (SCENARIO == 3 || SCENARIO == 4) ? 0 : 1;
const int PAYLOAD_SIZE = (SCENARIO == 3 || SCENARIO == 5) ? 20 : 500;

// ================= VARIABEL =================
WiFiClient net;
MQTTClient client(1024);   // buffer cukup untuk payload 500 byte

String clientId;
String topicTest;

unsigned long waktuMulai;
unsigned long waktuSelesai;

int berhasil = 0;
int gagal = 0;

// Dipakai callback untuk mendeteksi pesan yang kembali dari broker
volatile int expectedSeq = -1;
volatile bool pesanKembali = false;
volatile unsigned long tKembali = 0;

// ================= CALLBACK =================
void messageReceived(String &topic, String &payload) {
  // Format payload: "seq=<n>;AAAA..."
  int p = payload.indexOf(';');
  if (p < 5) return;

  int seqDiterima = payload.substring(4, p).toInt();

  // Hanya terima seq yang sedang ditunggu (pesan terlambat / duplikat diabaikan)
  if (seqDiterima == expectedSeq && !pesanKembali) {
    tKembali = millis();
    pesanKembali = true;
  }
}

// ================= FORMAT WAKTU =================
String getCurrentTime() {
  struct tm timeinfo;

  if (!getLocalTime(&timeinfo)) {
    return "Waktu tidak tersedia";
  }

  char buffer[25];
  strftime(buffer, sizeof(buffer), "%Y-%m-%d %H:%M:%S", &timeinfo);
  return String(buffer);
}

// ================= WIFI =================
void connectWiFi() {
  Serial.print("Menghubungkan WiFi");

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWiFi Terhubung");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
}

// ================= NTP WIB =================

// ================= MQTT CONNECT =================
bool connectMQTT() {
  if (client.connected()) return true;

  if (client.connect(clientId.c_str())) {
    client.subscribe(topicTest.c_str(), QOS);   // subscribe ke topik sendiri
    return true;
  }
  return false;
}

// ================= FORMAT TABEL =================
void printHeader() {
  Serial.println();
  Serial.println("========================================================================================================");
  Serial.println("skenario seq    waktu_mulai          payload  t_kirim  t_terima  rtt  status  rssi");
  Serial.println("========================================================================================================");
}

// ================= PENGUJIAN MQTT =================
void runScenario() {

  berhasil = 0;
  gagal = 0;

  // ID & topik unik agar tidak bentrok di broker publik
  clientId  = "esp32-uts-s" + String(SCENARIO) + "-" + String((uint32_t)ESP.getEfuseMac(), HEX);
  topicTest = "uts/iot/" + clientId;

  client.begin(MQTT_BROKER, MQTT_PORT, net);
  client.setKeepAlive(60);
  client.setTimeout(TIMEOUT);
  client.onMessage(messageReceived);

  Serial.print("Menghubungkan MQTT");
  while (!connectMQTT()) {
    Serial.print(".");
    delay(500);
  }
  Serial.println("\nMQTT Terhubung");

  waktuMulai = millis();

  String waktuMulaiPengujian = getCurrentTime();

  Serial.println("\n========================================");
  Serial.print  ("       PENGUJIAN MQTT QoS ");
  Serial.println(QOS);
  Serial.println("========================================");

  Serial.print("Skenario       : ");
  Serial.println(SCENARIO);

  Serial.print("Broker         : ");
  Serial.print(MQTT_BROKER);
  Serial.print(":");
  Serial.println(MQTT_PORT);

  Serial.print("Topik          : ");
  Serial.println(topicTest);

  Serial.print("QoS            : ");
  Serial.println(QOS);

  Serial.print("Jumlah pesan   : ");
  Serial.println(TOTAL_PESAN);

  Serial.print("Payload        : ");
  Serial.print(PAYLOAD_SIZE);
  Serial.println(" byte");

  Serial.print("Interval       : ");
  Serial.print(INTERVAL);
  Serial.println(" ms");

  Serial.print("Timeout        : ");
  Serial.print(TIMEOUT);
  Serial.println(" ms");

  Serial.print("Waktu mulai    : ");
  Serial.println(waktuMulaiPengujian);

  printHeader();

  // ================= LOOP PENGUJIAN =================
  for (int seq = 1; seq <= TOTAL_PESAN; seq++) {

    // Menunggu waktu kirim (sambil menjaga koneksi MQTT)
    unsigned long targetTime =
      waktuMulai + ((unsigned long)(seq - 1) * INTERVAL);

    while ((long)(millis() - targetTime) < 0) {
      client.loop();
      delay(1);
    }

    String waktuMulaiPesan = getCurrentTime();
    unsigned long tKirim = millis() - waktuMulai;

    // ================= PAYLOAD =================
    // "seq=<n>;" + padding 'A' sampai tepat PAYLOAD_SIZE byte
    String body = "seq=" + String(seq) + ";";
    while ((int)body.length() < PAYLOAD_SIZE) {
      body += "A";
    }
    int contentLength = body.length();

    unsigned long tTerima = 0;
    unsigned long rtt = 0;
    String terimaStr = "-";
    String rttStr = "-";
    bool sukses = false;

    // ================= PENGIRIMAN =================
    if (WiFi.status() == WL_CONNECTED && connectMQTT()) {

      expectedSeq = seq;
      pesanKembali = false;

      unsigned long requestStart = millis();

      bool terkirim = client.publish(topicTest.c_str(), body, false, QOS);

      // Tunggu pesan kembali dari broker (maks. TIMEOUT)
      if (terkirim) {
        while (!pesanKembali && (millis() - requestStart) < TIMEOUT) {
          client.loop();
          delay(1);
        }
      }

      if (terkirim && pesanKembali) {
        rtt = tKembali - requestStart;
        tTerima = tKembali - waktuMulai;

        terimaStr = String(tTerima);
        rttStr = String(rtt);

        sukses = true;
        berhasil++;
      } else {
        gagal++;
      }

    } else {
      gagal++;
    }

    int status = sukses ? 1 : 0;
    int rssi = WiFi.RSSI();

    // ================= OUTPUT TABEL =================
    Serial.printf(
      "%-8d %-6d %-20s %-8d %-9lu %-10s %-5s %-8d %d\n",
      SCENARIO,
      seq,
      waktuMulaiPesan.c_str(),
      contentLength,
      tKirim,
      terimaStr.c_str(),
      rttStr.c_str(),
      status,
      rssi
    );
  }

  // ================= AKHIR PENGUJIAN =================
  client.disconnect();

  waktuSelesai = millis();

  String waktuAkhirPengujian = getCurrentTime();

  // ================= REKAP =================
  float pdr = (float)berhasil / TOTAL_PESAN * 100.0;

  Serial.println();
  Serial.println("========================================");
  Serial.println("           HASIL PENGUJIAN");
  Serial.println("========================================");

  Serial.print("Skenario             : ");
  Serial.println(SCENARIO);

  Serial.print("Waktu mulai          : ");
  Serial.println(waktuMulaiPengujian);

  Serial.print("Waktu selesai        : ");
  Serial.println(waktuAkhirPengujian);

  Serial.print("Total pengujian      : ");
  Serial.println(TOTAL_PESAN);

  Serial.print("Payload              : ");
  Serial.println(PAYLOAD_SIZE);

  Serial.print("QoS                  : ");
  Serial.println(QOS);

  Serial.print("Berhasil             : ");
  Serial.println(berhasil);

  Serial.print("Gagal                : ");
  Serial.println(gagal);

  Serial.print("PDR                  : ");
  Serial.print(pdr, 2);
  Serial.println("%");

  Serial.print("Durasi pengujian     : ");
  Serial.print(waktuSelesai - waktuMulai);
  Serial.println(" ms");

  Serial.println("========================================");
}
